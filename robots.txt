# Account/transactional pages are intentionally NOT disallowed here.
# Each one already sends its own <meta name="robots" content="noindex, ...">
# (see cart.php, checkout.php, cart1.php, wishlist.php, login.php,
# register.php, profile.php, order_details.php, invoice.php,
# return_request.php). Blocking them here too would prevent Googlebot
# from ever crawling the page to see that noindex tag, which is exactly
# what causes the "Indexed, though blocked by robots.txt" warning in
# Search Console. Per Google's own guidance: use noindex OR robots.txt
# disallow to keep a URL out of the index, never both on the same URL.
# logout.php is the one exception below — it never renders a page (pure
# redirect + session destroy), so a noindex meta tag isn't possible and
# blocking it via robots.txt is the correct, non-conflicting choice.
Disallow: /logout
